#!/usr/bin/env python3
"""Verify this evidence pack on any machine with Python 3. No dependencies.

Checks, in order:
  1. the document file's sha256 is the one the run recorded;
  2. the canonical sections file's sha256 is the one the run recorded, and the guidance file's, when the
     run had guidance;
  3. every span the workbench marked verified is found again: the characters at the stored
     offsets of the stored section text match the quoted passage under the same ladder the
     workbench used (exact, normalized, casefold, typed tokens; letters-and-digits on records made
     before verifier v5; label stripping when the workbench recorded it). For a run verified under v7 or
     later the typed check reads tokens with the workbench's own pattern, copied into this file, and
     compares words whole, as the workbench did; for an earlier record it joins neighbouring words, as v5
     did, so it holds for records of v5 and v6 and is never stricter than what they recorded. A typed
     match needs at least one word or number: punctuation alone is never found;
  4. spans the workbench marked unverified are listed, with nothing to check: the workbench
     withheld them;
  5. when the pack carries the rebuilt model input, its hash is the one the run recorded when it
     made the call, and every context slice (the part of a section the model was shown) hashes
     to what run.json says and appears verbatim in the user message. The pack carries it only
     when it is the input of the recorded answer: an answer given in a later validation round
     (to a corrected message, the first plus a note on why the first answer was rejected) has
     an input the run did not keep, which is not reconstructable; this checker then says so and
     prints no PASS for the model input;
  6. a located span that lies beyond the slice the model was shown is reported: the quote exists
     in the document, but the model never saw those characters (a NOTE, not a failure). From
     verifier v5 the workbench searched only that window, so for such a run a span that ends
     beyond it fails;
  7. when the pack carries the model's output (model_output.txt), its bytes hash to what the run recorded
     when the model answered: the findings' sentences are that output with section handles written as
     labels, and this file keeps the model's own words.
For a run the workbench verified under verifier v8 or later (run.json options.verifier_version), check 3
also holds what v8 holds: neither edge of a span splits a run of letters and digits or a number as it
is written (its sign, parentheses, groups, currency and percent signs), and the typed check compares a
number with its sign, its parentheses and its written precision ("$15.00" is not "$15"). How a number is
written is the workbench's own definition, copied into this file when the pack was made; it needs
Python 3.11 or later, and an older Python fails these spans rather than pass them unchecked. A run
verified under an earlier verifier is checked as before, as its verifier recorded it.
For a run verified under verifier v9 or later, an amount written with a magnitude letter ("$5M", "USD 10K",
"2.5bn") and an amount in any currency (every Unicode currency sign, every ISO 4217 code in capitals,
"US$500") are numbers too, read as a reader sees the text (an invisible mark parts no sign from its number):
the typed check compares them with their sign, parentheses and currency, so "Adjustment Amount $5M" is not
"Adjustment Amount -$5M"; no edge falls inside one; and inside a number written with a sign or parentheses
no edge falls at a space either ("10K" is not inside "-USD 10K"). A v8 run is still checked as v8 read it.
For a run verified under verifier v10 or later, a short label after a minus sign or inside an opening parenthesis
("-Ksh 5M", "(RM5M)", "(Rp 5M)") is part of the number written after it, whatever currency it names, and a word
holding digits inside such a number ("RM5M") is part of its value. A v9 run is still checked as v9 read it.
For a run verified under verifier v11 or later, a comparator, a plus or plus-minus sign and a decimal point written
against a number are part of it (">$1,000,000", "≥99.9%", "±5%", "+/-5%", ".5%"), and digits joined by "/" or ":" are
one number ("2/3" is not "2-3"); the typed check compares the end of each sentence, so a quote cannot run one sentence
into the next; neither it nor the casefold check forgives the case of a clause's letter ("2(A)" is not "2(a)"); no
edge splits a word at a combining mark or a joiner; and a quote with no letter or digit is never found. A v10 run is
still checked as v10 read it.
Since the stress test of 2026-10-30 every field of findings.json, relations.json and run.json is either checked or
printed UNCHECKED, with why. run.json carries the run's stage rows as the run wrote them (stages: what each stage read
and produced, as hashes), and this file recomputes each one from the files here:
  8. reading: the document's hash and the sections' text; finding evidence: the question with the guidance's hash,
     and which sections were handed to the model, in order (run.json candidates); checking: the model's input and its
     answer; verifying: every span's section, offsets, verified flag and tier;
  9. the run's last stage, for a run that finished from 2026-10-30 on: the record it finished on, that is each
     finding's id, status, what set it and why, its relations and match counts, the relations the run located, its
     options (the verifier version and the window among them), and the run's identity, model and retrieval;
 10. each finding's words (topic, conclusion, the model's reading, every quote and the handle it cited) are the
     model's answer in model_output.txt with its section handles written as the sections' labels, and the question
     is the one in the model's input;
 11. a pointer emptied beside a hash the run recorded is a FAIL: a check the record says applies cannot be switched
     off by emptying the field that names its file. Only the original document's bytes may be truly absent, and
     their absence is printed UNCHECKED.
What no hash covers is printed UNCHECKED: a person's review, what the build that made the pack worked out from the
record (decided_by, flagged, concerns, clause_label), when the run ran and was packed and, on a run finished before
2026-10-30, its statuses, relations and options.
This verifier establishes integrity between the original input, the canonical reading, the context
given to the model, the model's answer and the stored findings. It does not, and cannot, say the conclusions are
right. The pack is not signed: it shows that its files agree with each other and with the stage hashes run.json
carries, not that nobody changed them together; compare run.json's stage rows with the run's in the workbench.
Exit code 0 when every check passes, 1 when one fails, and 2 when the run was verified under a verifier this file
does not know: it then judges nothing (NOT CHECKED).
"""

def sha256(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def sha256_text(text):
    return hashlib.sha256(text.encode("utf-8")).hexdigest()


def file_hash(here, name):
    """The sha256 of a file run.json names, or None when the pack has no such file."""
    path = here / name
    return sha256(path) if path.is_file() else None


def holds(here, name, expected):
    """Whether the pack holds the file run.json names and it hashes to `expected`: a file that is not there matches
    nothing, not even a hash that is null."""
    found = file_hash(here, name)
    return found is not None and found == expected


def normalize(text, version=0):
    table = quote_map(version)
    out = []
    pending = False
    for ch in text:
        mapped = table.get(ch, ch)
        if mapped == "":
            continue
        if mapped.isspace():
            pending = True
            continue
        if pending and out:
            out.append(" ")
        pending = False
        out.append(mapped)
    return "".join(out)


def alnum(text):
    return "".join(ch.casefold() for ch in text if ch.isalnum())


TOKEN = re.compile(
    r"(?P<section>\d+(?:\.\d+){2,}(?:\([a-z0-9]+\))*|\d+(?:\.\d+)?(?:\([a-z0-9]+\))+)"
    r"|(?P<currency>(?:[$€£¥]|USD|EUR|GBP|CHF|CAD|AUD)\s?\d+(?:,\d+)*(?:\.\d+)?)"
    r"|(?P<percent>\d+(?:,\d+)*(?:\.\d+)?\s?(?:%|percent\b|per\s+cent\b))"
    r"|(?P<number>\d+(?:,\d+)*(?:\.\d+)?)"
    r"|(?P<word>[^\W\d_]+)"
    r"|(?P<punct>[^\w\s]|_)",
    re.IGNORECASE,
)


def number(text, precise=False):
    compact = text.replace(" ", "")
    # A comma separates thousands only in thousands grouping: "1,5" and "15,00" are not 15 and 1500 (verifier v7).
    if "," in compact and not re.fullmatch(r"\d{1,3}(?:,\d{3})+(?:\.\d+)?", compact):
        return compact
    if precise:
        # v8: the digits as written, so "15.00" is not "15" (verify/tokens.canonical_number).
        cleaned = compact.replace(",", "")
        try:
            decimal.Decimal(cleaned)
        except decimal.InvalidOperation:
            return text
        return cleaned
    try:
        return format(decimal.Decimal(compact.replace(",", "")).normalize(), "f")
    except decimal.InvalidOperation:
        return text


NUMBER_KINDS = ("number", "currency", "percent", "section")


def typed_tokens(text, version=0):
    """The workbench's typed tokens (verify/tokens.py) in standard-library form, each [kind, canonical, start, end]. For
    a run verified under v7 or later, its own pattern: no token ends inside a run of letters and digits, and a number
    keeps its written precision. Under v8 a number also carries its sign and parentheses, and starts and ends where it
    is written (verify/tokens.signed). Under v11 a number whose digits "/" or ":" join is one token, and the end of each
    sentence is a token of kind "break" (verify/tokens.compared). For an earlier run, the pack's reading before v7."""
    v7, v8 = version >= 7, version >= 8
    tokens = []
    for match in (TOKEN_V7 if v7 else TOKEN).finditer(text):
        kind = match.lastgroup
        surface = match.group(0)
        if kind == "word":
            canonical = surface.casefold()
        elif kind == "number":
            canonical = number(surface, v7)
        elif kind == "currency":
            symbol = re.match(r"(?:[$€£¥]|USD|EUR|GBP|CHF|CAD|AUD)", surface, re.IGNORECASE).group(0)
            canonical = symbol.upper() + number(surface[len(symbol):], v7)
        elif kind == "percent":
            canonical = number(re.match(r"[\d,]*(?:\.\d+)?", surface).group(0), v7) + "%"
        elif kind == "section":
            canonical = surface.lower()
        else:
            canonical = surface
        tokens.append([kind, canonical, match.start(), match.end()])
    if v8:
        # v9 (verify/tokens.signed): an amount word is a number too, what stands before a number is compared without
        # spaces or case, and a currency written as letters before it belongs to it.
        v9, v10, v11 = version >= 9, version >= 10, version >= 11
        hidden = deleted(version)

        def amount(t):
            if not v9 or t[0] != "word":
                return False
            if AMOUNT_WORD.fullmatch(text, t[2], t[3]) is not None:
                return True
            return v10 and any(c.isdigit() for c in text[t[2]:t[3]])  # v10: a word holding digits, inside a number

        def value(t):
            if not amount(t):
                return t[1]
            parts = AMOUNT_WORD.fullmatch(text, t[2], t[3])
            if parts is not None:
                return (parts.group("code") or "").upper() + parts.group("digits") + (parts.group("magnitude") or "").casefold()
            parts = LABELLED_WORD.fullmatch(text, t[2], t[3])
            if parts is None:
                return t[1]
            return parts.group(1).upper() + parts.group(2) + (parts.group(3) or "").casefold()

        numbers = [i for i, t in enumerate(tokens) if t[0] in NUMBER_KINDS or amount(t)]
        k = 0
        # v9 reads where a number is written as a reader sees the text (characters normalisation deletes left out).
        spans = written_numbers(text, version) if v9 else [m.span() for m in WRITTEN.finditer(text)]
        for low, high in spans:
            while k < len(numbers) and tokens[numbers[k]][3] <= low:
                k += 1
            inside = []
            while k < len(numbers) and tokens[numbers[k]][2] < high:
                inside.append(numbers[k])
                k += 1
            if not inside:
                continue
            first, last = tokens[inside[0]], tokens[inside[-1]]
            before = "".join("-" if c in MINUS_SIGNS else c for c in text[low:first[2]])
            after = text[last[3]:high]
            head, tail = value(first), value(last)
            if v11:  # a clause's letter keeps its case: "2(A)" is not "2(a)"
                head = text[first[2]:first[3]] if first[0] == "section" else head
                tail = text[last[2]:last[3]] if last[0] == "section" else tail
            if v9:
                before = "".join(c for c in before if c not in hidden and not c.isspace()).upper()
                after = "".join(c for c in after if c not in hidden)
                j = inside[0] - 1
                while j >= 0 and tokens[j][2] >= low:
                    if tokens[j][0] == "word":
                        tokens[j][0] = "punct"
                    j -= 1
            if v11 and len(inside) > 1 and any(c in JOINED for c in text[first[3]:last[2]]):
                # One value, its joints included, a space before a fraction kept as one (verify/tokens.signed).
                joined, previous = head, first
                for t in tokens[inside[0] + 1:inside[-1] + 1]:
                    gap = "".join(c for c in text[previous[3]:t[2]] if c not in hidden)
                    joined += (" " if any(c.isspace() for c in gap) else "") + "".join(c for c in gap if not c.isspace())
                    joined += tail if t is last else (value(t) if t[0] == "word" else text[t[2]:t[3]] if t[0] == "section" else t[1])
                    previous = t
                first[1:] = [before + joined + after, low, high]
                for t in tokens[inside[0] + 1:inside[-1] + 1]:
                    t[0] = "punct"
            elif len(inside) == 1:
                first[1:] = [before + head + after, low, high]
            else:
                first[1:3] = [before + head, low]
                last[1], last[3] = tail + after, high
    if version >= 11:
        tokens = whole_words(text, tokens)
        choices = [["symbol", ch, at, at + 1] for at, ch in enumerate(text) if ch in CHOICE_MARKS]
        tokens = sorted(tokens + [["break", "", at, at + 1] for at in sentence_ends(text)] + choices, key=lambda t: t[2])
    return tokens


def caseless(text):
    """Unicode's canonical caseless form, composed (verify/tokens.caseless, v11)."""
    return unicodedata.normalize("NFC", unicodedata.normalize("NFD", text).casefold())


def marked(ch):
    """A combining mark or a joiner: part of the letters it is written in (verify/tokens.marked, v11)."""
    return ch in JOINERS or unicodedata.category(ch)[0] == "M"


def whole_words(text, tokens):
    """v11's words, read whole through their marks and joiners and compared caseless; a clause's letter alone in
    parentheses keeps its case (verify/tokens.whole_words)."""
    out = []
    k = 0
    while k < len(tokens):
        token = tokens[k]
        k += 1
        if token[0] != "word":
            out.append(token)
            continue
        end = token[3]
        while end < len(text) and marked(text[end]):
            end += 1
            while end < len(text) and (text[end].isalnum() or marked(text[end])):
                end += 1
        while k < len(tokens) and tokens[k][2] < end:
            k += 1
        surface = text[token[2]:end]
        if end > token[3] or token[1] == surface.casefold():
            if text[token[2] - 1:token[2]] == "(" and CLAUSE_LETTER.fullmatch(text, token[2] - 1, end + 1):
                out.append(["word", surface, token[2], end])
            else:
                out.append(["word", caseless(surface), token[2], end])
        else:
            out.append(token)
    return out


def typed(text, version=0):
    """What the typed check compares: the significant tokens, words whole for a run verified under v7 or later (as the
    workbench compares them since v6); for an earlier run, runs of joined words, never stricter than v5 and v6."""
    items = []
    run = []
    for kind, canonical, _, _ in typed_tokens(text, version):
        if kind == "punct":
            continue
        if kind == "word" and version < 7:
            run.append(canonical)
            continue
        if run:
            items.append(("word", "".join(run)))
            run = []
        items.append((kind, canonical))
    if run:
        items.append(("word", "".join(run)))
    return items


def seen(text, at, step, version=8):
    """The two characters a reader sees before (step -1) or after (step 1) the edge before text[at] (verify/spans._seen)."""
    hidden = deleted(version)
    near = text[max(0, at - 2):at] if step < 0 else text[at:at + 2]
    if hidden.isdisjoint(near):
        return near
    found = []
    i = at - 1 if step < 0 else at
    while 0 <= i < len(text) and len(found) < 2:
        if text[i] not in hidden:
            found.append(text[i])
        i += step
    return "".join(reversed(found)) if step < 0 else "".join(found)


def written_numbers(text, version=8):
    """Where each number of text lies as it is written, as the reader sees it (verify/spans._numbers)."""
    hidden = deleted(version)
    shown = [i for i, ch in enumerate(text) if ch not in hidden]
    visible = "".join(text[i] for i in shown)
    pattern = WRITTEN_V11 if version >= 11 else WRITTEN_V10 if version >= 10 else WRITTEN_V9 if version >= 9 else WRITTEN
    return [(shown[m.start()], shown[m.end() - 1] + 1) for m in pattern.finditer(visible)]


def sentence_ends(text):
    """Where a sentence ends, read as the reader sees the text (verify/tokens.sentence_ends, v11)."""
    shown = [i for i, ch in enumerate(text) if ch not in DELETED_V11]
    visible = "".join(text[i] for i in shown)
    return [shown[m.start()] for m in SENTENCE_END.finditer(visible) if ends_sentence(visible, m)]


def ends_sentence(visible, m):
    """A semicolon always; after a space, a letter not in lower case; with no space, a capital between two lower-case
    letters (verify/tokens._ends_sentence)."""
    if m.group("next") is not None:
        return not m.group("next").islower()
    if m.group("tight") is not None:
        at = m.start("tight")
        return m.group("tight").isupper() and visible[m.start() - 1:m.start()].islower() and visible[at + 1:at + 2].islower()
    return True


def in_run(ch, version):
    """A letter or a digit; from v11 also a combining mark or a joiner, which belong to their word (verify/spans._in_run)."""
    return ch.isalnum() or (version >= 11 and marked(ch))


def splits(text, at, numbers, version=8):
    """Whether the edge before text[at] falls inside a run of letters and digits or a number as written (verify/spans._splits)."""
    before, after = seen(text, at, -1, version), seen(text, at, 1, version)
    if not before or not after:
        return False
    b, a = before[-1:], after[:1]
    marks = DIGIT_MARKS_V11 if version >= 11 else DIGIT_MARKS
    if (in_run(b, version) and in_run(a, version)) or (b.isdigit() and a in marks and after[1:2].isdigit()) or (b in marks and before[-2:-1].isdigit() and a.isdigit()):
        return True
    if version >= 11 and ((b.isdigit() and a == " " and fraction(after[1:2])) or (b == " " and before[-2:-1].isdigit() and fraction(a))):
        return True
    attached = ATTACHED_V11 if version >= 11 else ATTACHED_V9 if version >= 9 else ATTACHED
    if b in attached or a in attached:
        return any(low < at < high for low, high in numbers)
    if version < 9:
        return False
    # v9: anywhere inside a number written with a sign or parentheses ("-USD 10K", "-Rs. 5M"); v11: or opened by a
    # comparator, a plus sign or a decimal point (">$1,000,000", ".5%").
    openers = SIGN_OPENERS_V11 if version >= 11 else SIGN_OPENERS
    # v11: nor after a number's first digit ("1/2%" is not inside "1 1/2%").
    return any(low < at < high and (text[low] in openers or (version >= 11 and any(c.isdigit() for c in text[low:at]))) for low, high in numbers)


def fraction(ch):
    """A vulgar fraction, which a space after a whole number joins to it (verify/tokens._fraction, v11)."""
    return len(ch) == 1 and unicodedata.category(ch) == "No" and 0 < unicodedata.numeric(ch, 0) < 1


def edges_hold(text, start, end, version=8):
    numbers = written_numbers(text, version)
    return not splits(text, start, numbers, version) and not splits(text, end, numbers, version)


def verifier_number(run):
    """The verifier the run recorded (options.verifier_version): 0 for a run recorded before versions were, None for a
    name this file cannot read."""
    named = (run.get("options") or {}).get("verifier_version")
    if not named:
        return 0
    found = re.fullmatch(r"v(\d+)", str(named))
    return int(found.group(1)) if found else None


def window_of(options):
    """The window a run searched each section in: its recorded option, or the window of every run recorded before the
    option existed (analysis/service.window_of)."""
    value = options.get("section_window", LEGACY_WINDOW)
    return int(value) if isinstance(value, (int, float, str)) else LEGACY_WINDOW


def follows(names):
    """Whether stage rows are a way through the run's transition table from its creation (runs/transitions.ALLOWED)."""
    at = "created"
    for name in names:
        if name not in TRANSITIONS.get(at, ()):
            return False
        at = name
    return True


def located(findings):
    """What the verifying stage hashed: each span's section, offsets, verified flag and tier, in finding order
    (runs/service.execute_run)."""
    spans = [[s["section_id"], s["start"], s["end"], s["verified"], s["method"]] for f in findings for s in f["spans"]]
    return sha256_text(json.dumps(spans, sort_keys=True))


def record_of(run, findings, relations):
    """The record the run finished on, as the workbench hashed it when the run ended (runs/service.finished_record):
    what no earlier stage's hash covers. `relations` is relations.json as read, or None when the pack has none."""
    record = {
        "run": run["run_id"],
        "fingerprint": run["fingerprint"],
        "document": run["document"]["id"],
        "task": run["task"],
        "provider": run["provider"],
        "model": run["model"],
        "prompt_version": run["prompt_version"],
        "prompt_hash": run["prompt_hash"],
        "retrieval_mode": run["retrieval"]["mode"],
        "candidates": run["candidates"],
        "options": run["options"],
        "stage": run["stage"],
        "reason": run["reason"],
        "relations": relations,
        "findings": [
            {
                "id": f["id"],
                "ordinal": f["ordinal"],
                "status": f["status"],
                "status_source": f["status_source"],
                "status_reason": f["status_reason"],
                "relations": f["relations"],
                "match_counts": [s["match_count"] for s in f["spans"]],
            }
            for f in findings
        ],
    }
    return sha256_text(json.dumps(record, sort_keys=True))


def slices_of(candidates, sections, window):
    """The part of each candidate section the model was shown, in rank order (application/reconstruct_input)."""
    slices = []
    for rank, candidate in enumerate(candidates, start=1):
        text = sections[candidate["section_id"]]["text"]
        end = min(len(text), window)
        slices.append(
            {"label": candidate["label"], "section_id": candidate["section_id"], "rank": rank, "start": 0, "end": end, "truncated": len(text) > window, "sha256": sha256_text(text[:end])}
        )
    return slices


def labels_of(candidates, sections):
    """What each handle the model was given reads as once its prose reaches a person: "§" and the section's number, or
    its heading when it has none (runs/service.execute_run)."""
    labels = {}
    for candidate in candidates:
        section = sections[candidate["section_id"]]
        labels[candidate["label"]] = f"§{section['number']}" if section["number"] else section["heading"]
    return labels


def label_handles(text, labels):
    """The model's sentence as the workbench stores it: each sec_N handle written as its section's label, and one that
    names no section it was handed, or carries a sub-reference, as UNRESOLVED (runs/prose.label_handles)."""
    if text is None:
        return None

    def label(match):
        if match.group("bracketed_sub") or match.group("bare_sub"):
            return UNRESOLVED
        return labels.get("sec_" + (match.group("bracketed") or match.group("bare"))) or UNRESOLVED

    cleaned = SPACE_BEFORE_STOP.sub(r"\1", EMPTY_BRACKETS.sub("", HANDLE.sub(label, text)))
    return SPACES.sub(" ", cleaned).strip()


def proposed(answer):
    """The findings the model's answer proposes, as the run made findings of them: none when it says the sections hold
    no answer, or when it is not JSON (runs/service.execute_run)."""
    try:
        parsed = json.loads(answer)
    except ValueError:
        return []
    if not isinstance(parsed, dict) or parsed.get("insufficient_evidence") or not parsed.get("findings"):
        return []
    return parsed["findings"]


def differences(finding, ordinal, item, labels):
    """The fields of one finding that are not the model's item as the workbench stored it: its handles written as the
    sections' labels, its topic cut to 255 characters, each quote and cited handle as given (runs/service)."""
    different = []
    expected = {"ordinal": ordinal, "topic": label_handles(item["topic"], labels)[:255], "conclusion": label_handles(item["conclusion"], labels)}
    for name in ("observed", "required", "guidance_reference", "suggested_position"):
        expected[f"model.{name}"] = label_handles(item.get(name), labels) or None
    shown = {**finding, **{f"model.{name}": value for name, value in finding["model"].items()}}
    different += [name for name, value in expected.items() if shown[name] != value]
    if finding["shown_conclusion"] != (NOT_FOUND if finding["status"] == "missing" else finding["conclusion"]):
        different.append("shown_conclusion")
    if len(finding["spans"]) != len(item["evidence"]):
        different.append("spans")
    for k, (span, quoted) in enumerate(zip(finding["spans"], item["evidence"])):
        different += [f"span {k + 1} {name}" for name, value in (("ordinal", k), ("quote", quoted["quote"]), ("cited_label", quoted["section_id"])) if span[name] != value]
    return different



def typed_at(text, start, end, version):
    """The significant tokens of the text the workbench searched, read in place as it read them, that make up
    text[start:end]; None unless the span begins and ends at the edges of such tokens, as every span its typed tier
    reports does (verify/tokens.locate_tokens). Read in place, not as a cut-out: a "$" after a digit is no currency
    sign to the workbench, and a minus sign is read as the reader sees it."""
    inside = [t for t in typed_tokens(text, version) if t[0] != "punct" and start <= t[2] and t[3] <= end]
    if not inside or inside[0][2] != start or inside[-1][3] != end:
        return None
    return [(t[0], t[1]) for t in inside]


def normalize_with_map(text):
    """The normalised text and each of its characters' index in the original (verify/spans.normalize_with_map)."""
    out, index_map, pending = [], [], False
    for i, ch in enumerate(text):
        mapped = QUOTE_MAP_V11.get(ch, ch)
        if mapped == "":
            continue
        if mapped.isspace():
            pending = True
            continue
        if pending and out:
            out.append(" ")
            index_map.append(index_map[-1] + 1 if index_map else i)
        pending = False
        out.append(mapped)
        index_map.append(i)
    return "".join(out), index_map


def alnum_with_map(text):
    """Letters and digits casefolded, a decimal point between digits kept, each mapped to its index (verify/spans)."""
    out, index_map, last = [], [], len(text) - 1
    for i, ch in enumerate(text):
        if ch.isalnum():
            folded = ch.casefold()
            out.append(folded)
            index_map.extend([i] * len(folded))
        elif ch == "." and 0 < i < last and text[i - 1].isdigit() and text[i + 1].isdigit():
            out.append(ch)
            index_map.append(i)
    return "".join(out), index_map


def strip_label(quote, label):
    """The quote without the section label copied in front of it, or None (verify/spans.strip_label, v11)."""
    first, _, rest = label.partition(" ")
    number, heading = (first, rest) if re.fullmatch(r"\d+(?:\.\d+)*", first) else ("", label)
    norm_quote, norm_map = normalize_with_map(quote)
    start = 0
    if number:
        written = re.match(re.escape(number) + r"[.):\]]?(?:\s|$)", norm_quote)
        if written is not None:
            if written.end() >= len(norm_quote):
                return None
            start = norm_map[written.end()]
    heading_alnum, _ = alnum_with_map(heading)
    if len(heading_alnum) < 2:
        return None
    body = quote[start:]
    body_alnum, body_map = alnum_with_map(body)
    if not body_alnum.startswith(heading_alnum) or len(body_alnum) == len(heading_alnum):
        return None
    begin, heading_end = body_map[len(heading_alnum)], body_map[len(heading_alnum) - 1]
    separator = body[heading_end + 1:heading_end + 2]
    if separator and not (separator.isspace() or separator in LABEL_ENDS):
        return None  # a heading copied whole ends where a separator follows it
    begin = next((low for low, high in (m.span() for m in WRITTEN_V11.finditer(body)) if heading_end < low < begin < high), begin)
    remainder = body[begin:].lstrip()
    return remainder or None


def same(found, quote, method, version=0, where=None, label=""):
    if version >= 11 and not any(ch.isalnum() for ch in quote):
        return False  # v11: punctuation alone is never found
    base = method.split(":")[-1]
    if "unprefixed" in method:
        if version >= 11:
            # The label stripped as the workbench strips it, the rest compared under the span's own tier.
            rest = strip_label(quote, label)
            return rest is not None and same(found, rest, base, version, where)
        return bool(alnum(found)) and alnum(quote).endswith(alnum(found))
    if base == "exact":
        return found == quote
    if base == "normalized":
        return normalize(found, version) == normalize(quote, version).strip()
    if base == "casefold":
        # v11: case is forgiven, never a clause's letter ("Section 2(A)" is not found at "Section 2(a)").
        return normalize(found, version).casefold() == normalize(quote, version).strip().casefold() and (version < 11 or CLAUSE_LETTER.findall(found) == CLAUSE_LETTER.findall(quote))
    if base == "alnum":
        return alnum(found) == alnum(quote)
    if base == "typed":
        # From v7 the span is read where it stands in the searched text (where = text, start, end), as the workbench read it.
        items = typed_at(*where, version) if version >= 7 and where is not None else typed(found, version)
        return bool(items) and items == typed(quote, version)
    return False


def main(here=HERE):
    # A section's number or heading the console cannot encode (cp1252 when piped on Windows) is escaped, never fatal.
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(errors="backslashreplace")
    try:
        return check(here)
    except (KeyError, IndexError, TypeError, AttributeError, ValueError, OSError) as error:
        # A file edited out of the shape this checker reads (a field removed, a pointer to no file) fails, named, and is
        # never a traceback a reader could take for the checker's own fault.
        print(f"FAIL the pack's files are not in the shape this checker reads: {type(error).__name__} {error}")
        return 1


def check(here):
    run = json.loads(text_of(here / "run.json"))
    version = verifier_number(run)
    if version is None or version > KNOWN_VERIFIER:
        # A newer verifier's rules are not this file's: judged by its own, a run recorded as "v99" printed PASS (stress
        # test, 2026-10-30). Nothing is judged then, not even a hash: the reader is sent to a checker that knows them.
        named = run["options"]["verifier_version"]
        print(
            f"NOT CHECKED this run was verified under {named!r}, and this checker knows the verifier up to v{KNOWN_VERIFIER}: "
            f"it judges nothing it does not know. Check the pack with the verify.py of a Verity build that knows {named!r}."
        )
        return 2
    sections = {s["id"]: s for s in json.loads(text_of(here / "sections.json"))}
    findings = json.loads(text_of(here / "findings.json"))
    failures = left = 0

    def report(ok, label):
        nonlocal failures
        failures += 0 if ok else 1
        print(("PASS " if ok else "FAIL ") + label)

    def unchecked(fields, why):
        """A field nothing here can check, said so: `<file> <field>, <field>; <file> <field>`, then why."""
        nonlocal left
        left += 1
        print(f"UNCHECKED {fields}: {why}")

    # The run's record. run.json carries the stage rows the run wrote (stages), each with the hash of what the stage
    # read and produced, and every one is recomputed below from the files here. Until 2026-10-30 the pack carried none,
    # and findings.json and most of run.json were bound to nothing (stress test, 2026-10-30).
    stages = run.get("stages")
    bound = stages is not None
    rows = {row["stage"]: row for row in stages} if bound else {}
    order = [row["stage"] for row in stages] if bound else []
    # Every stage of a run written since 2026-09-28 hashed what it read and produced; a run older than that hashed nothing.
    era = any(row["input_hash"] or row["output_hash"] for row in stages) if bound else False
    if bound:
        # Every hash a row carries is one some check below recomputes: a hash on a row no stage writes it on is no record.
        stray = [row["stage"] for row in stages if (row["input_hash"] is not None and row["stage"] == "failed") or (row["output_hash"] is not None and row["stage"] in ENDINGS)]
        report(
            follows(order) and order[-1:] == [run["stage"]] and len(rows) == len(order) and not stray,
            f"stage rows: {' > '.join(order)}, a way through the run's stages that ends where run.json says it ended",
        )
    elif run.get("run_id"):
        report(False, f"stage rows: run.json names run {run['run_id']} but carries none of its stage rows (stages); every pack made with this checker carries them")
    else:
        unchecked(
            "run.json every field; findings.json every field; relations.json every field",
            "this run.json names no run and carries no stage rows, so the files below are checked against each other, not against what a run recorded",
        )
    model_input = run.get("model_input") or {}
    output = run.get("model_output") or {}
    # A pack without stage rows (made by hand, as the tests of this file make them) is read from run.json's own copies.
    copies = {
        ("reading", "output_hash"): run.get("sections_text_sha256_recorded"),
        ("checking", "input_hash"): model_input.get("input_sha256_recorded"),
        ("checking", "output_hash"): output.get("sha256_recorded"),
    }

    def recorded(stage, which):
        return (rows.get(stage) or {}).get(which) if bound else copies.get((stage, which))

    if bound:
        # run.json's own copies are the rows' hashes, or the pack says two things about one record.
        for (stage, which), name in (
            (("reading", "output_hash"), "sections_text_sha256_recorded"),
            (("checking", "input_hash"), "model_input.input_sha256_recorded"),
            (("checking", "output_hash"), "model_output.sha256_recorded"),
        ):
            if copies[(stage, which)] != recorded(stage, which):
                report(False, f"run.json {name} is not the {which} its {stage} stage row carries")

    def required(stage, which):
        """Whether the run must have written this hash. On a run that wrote stage hashes at all, a stage records the hash
        of its input when it opens and of its output when it ends; it ends without an output hash only when it failed,
        or when verifying found no passage (runs/service.execute_run). The last stage's record is new on 2026-10-30,
        so it is never required: its absence is UNCHECKED."""
        if not era or stage not in rows:
            return False
        if which == "input_hash":
            return stage in ("reading", "finding_evidence", "checking", "verifying")
        later = order[order.index(stage) + 1 :]
        return bool(later) and later[0] != "failed" and not (stage == "verifying" and not findings)

    def against(stage, which, computed, label, binds, why=None):
        """One recorded stage hash against what the files here give. A hash the run recorded is checked, and is a FAIL
        when the pack does not carry what it covers: a pointer emptied beside it does not switch the check off (stress
        test, 2026-10-30: it turned into a NOTE that said "older record"). A hash the row lacks is a FAIL when the run
        must have written it; otherwise what it would bind is UNCHECKED."""
        value = recorded(stage, which)
        if value is None:
            if required(stage, which):
                report(False, f"{label}: the run wrote its stage hashes, but run.json's {stage} row carries no {which}")
            elif bound and computed is not None:
                unchecked(binds, why or f"run.json's {stage} stage row carries no {which} (stages.{which}), so nothing here is checked against it")
            return
        if computed is None:
            report(False, f"{label} {value[:16]}..., but the pack does not carry what that hash covers")
            return
        report(computed == value, f"{label} {value[:16]}...")

    # What the reading stage read and produced: the document, and the sections' text.
    document = run["document"]
    if bound:
        against("reading", "input_hash", document["sha256"], "reading: the document run.json names is the one the run read, recorded", "run.json document.sha256")
    if document.get("file"):
        report(holds(here, document["file"], document["sha256"]), f"document bytes: sha256 {str(document['sha256'])[:16]}...")
    else:
        unchecked(
            "run.json document.file",
            "the original document's bytes are not in this pack, so nothing here checks them against document.sha256; the reading stage's hash still ties the sections to what the run read",
        )
    report(holds(here, "sections.json", run["sections_sha256"]), f"canonical sections: sha256 {str(run['sections_sha256'])[:16]}...")
    joined = sha256_text(chr(31).join(s["text"] for s in sorted(sections.values(), key=lambda s: s["ordinal"])))
    against("reading", "output_hash", joined, "sections are what the run read: recorded", "sections.json text")
    guidance_file = run["guidance_file"] if bound else run.get("guidance_file")  # a pack made with this file names it, null or not
    guidance = text_of(here / guidance_file) if guidance_file and (here / guidance_file).is_file() else None
    if guidance_file:
        report(holds(here, guidance_file, run.get("guidance_sha256")), f"guidance: sha256 {str(run.get('guidance_sha256'))[:16]}...")
    elif run.get("guidance_sha256"):
        report(False, "guidance: run.json records the run's guidance (guidance_sha256) but the pack carries no guidance file (guidance_file)")

    # What finding evidence read and chose: the question with the guidance's hash, and the sections handed to the model.
    candidates = run.get("candidates")
    options = run.get("options") or {}
    if bound:
        asked = sha256_text(f"{run['question']}{chr(31)}{run['guidance_sha256'] or ''}")
        against("finding_evidence", "input_hash", asked, "question and guidance: what the run was asked with, recorded", "run.json question, guidance_sha256")
        chosen = sha256_text(chr(31).join(c["section_id"] for c in candidates)) if candidates is not None else None
        against("finding_evidence", "output_hash", chosen, "candidates: the sections handed to the model, in order, are the ones the run chose, recorded", "run.json candidates.section_id")
        for c in candidates or []:
            section = sections.get(c["section_id"])
            if section is None or (c["number"], c["heading"]) != (section["number"], section["heading"]):
                report(False, f"candidates: {c['label']} is not section {c['section_id']} of sections.json as run.json names it")
        # The window and the number of sections asked for are the options' (bound below by the run's record).
        window = window_of(options)
        said = {
            "verifier.window_chars": ((run.get("verifier") or {}).get("window_chars"), window),
            "model_input.prompt_window_chars": (model_input.get("prompt_window_chars"), window),
            "retrieval.requested_k": ((run.get("retrieval") or {}).get("requested_k"), options.get("retrieval_k")),
        }
        wrong = [name for name, (value, expected) in said.items() if value != expected]
        report(not wrong, f"window and retrieval: run.json's window ({window} characters) and the sections it asked for are its options'" + (f"; not {', '.join(wrong)}" if wrong else ""))
    else:
        # The run searched each section within its window (what the prompt carried); its edges are read the same way.
        window = (run.get("verifier") or {}).get("window_chars") or run.get("prompt_window_chars")

    # What checking was given and answered: the model's input, when the pack can carry it, and the model's answer.
    user = None
    if model_input.get("user_file") and model_input.get("system_file"):
        system = text_of(here / model_input["system_file"])
        user = text_of(here / model_input["user_file"])
        prompt_hex = sha256_text(system)
        against("checking", "input_hash", sha256_text(prompt_hex + user), "model input: rebuilt messages hash to what the run recorded", "run.json model_input")
        report(prompt_hex == run["prompt_hash"], f"prompt {run['prompt_version']}: system text hashes to the recorded prompt hash")
        given = f"LEGAL GUIDANCE: {guidance.strip()}" if guidance and guidance.strip() else "LEGAL GUIDANCE: none supplied"
        report(user.startswith(f"QUESTION: {str(run['question']).strip()}\n\n{given}\n\nCONTRACT: "), "question: run.json's question and the guidance are the ones the model was asked, in model_input/user.txt")
        if bound and model_input.get("reconstructable") is not True:
            report(False, "model input: run.json says the messages behind the answer cannot be rebuilt (model_input.reconstructable), yet carries them")
    else:
        if model_input:
            print(f"NOTE the model input could not be rebuilt from the record: {model_input.get('problem')}")
        if bound and model_input.get("reconstructable") is not False:
            report(False, "model input: run.json says the messages behind the answer are in this pack (model_input.reconstructable), but names no file for them")
        if recorded("checking", "input_hash"):
            unchecked("run.json model_input", f"the messages behind the answer are not in this pack, so the hash the run recorded when it called the model, {recorded('checking', 'input_hash')[:16]}..., is checked against nothing here")
    answer = text_of(here / output["file"]) if output.get("file") and (here / output["file"]).is_file() else None
    answered = file_hash(here, output["file"]) if output.get("file") else None
    against("checking", "output_hash", answered, "model output: the recorded answer hashes to what the run recorded", "run.json model_output.file")
    if bound and "verifying" in rows:
        against("verifying", "input_hash", answered, "verifying: the answer it read is the one recorded", "run.json model_output.file")

    # The model's words in findings.json are its answer, its handles written as the sections' labels (runs/prose).
    if bound and "verifying" in rows and candidates is not None and answer is not None:
        labels = labels_of(candidates, sections)
        items = proposed(answer)
        report(len(findings) == len(items), f"findings: {len(findings)}, as many as the model's answer in model_output.txt proposes ({len(items)})")
        for ordinal, (finding, item) in enumerate(zip(findings, items)):
            different = differences(finding, ordinal, item, labels)
            report(not different, f"finding {ordinal + 1}: its words are the model's answer in model_output.txt, its handles written as the sections' labels" + (f"; not {', '.join(different)}" if different else ""))
    elif bound and "verifying" not in rows and findings:
        report(False, "findings: findings.json holds findings, but the run never reached the verifying stage that makes them")
    # Where each span was located, and whether: what the verifying stage produced.
    if bound:
        spans = any(f["spans"] for f in findings)
        against("verifying", "output_hash", located(findings) if spans else None, "located spans: every span's section, offsets, verified flag and tier are what the run recorded", LOCATED_FIELDS)
    # The record the run finished on: what no earlier stage covers, hashed as its last stage's input from 2026-10-30.
    if bound:
        relations = json.loads(text_of(here / run["relations_file"])) if run.get("relations_file") else None
        fields = RECORD_FIELDS.format(prompt="" if user is not None else "prompt_hash, ")
        last = order[-1] if order else None
        if last in ("complete", "unresolved"):
            against(
                last,
                "input_hash",
                record_of(run, findings, relations),
                "the record the run finished on: each finding's status and what set it, the relations, the options, the model and the run's identity hash to what its last stage recorded",
                fields,
                f"run.json's {last} stage row carries no hash of the record the run finished on (stages.input_hash); a run that finished before 2026-10-30 recorded none, so its statuses, relations and options are the pack's word only",
            )
        else:
            unchecked(fields, "the run did not finish its stages, so it recorded no hash of a finished record; its statuses, relations and options are the pack's word only")

    # The part of each candidate the model was shown: worked out here from the candidates, the sections and the window.
    if bound:
        slices = slices_of(candidates, sections, window) if candidates is not None else []
        copy = run.get("context_slices")
        returned = (run.get("retrieval") or {}).get("returned")
        report((copy == slices if copy else user is None) and returned == (len(copy) if copy else None), "context slices: run.json's are the parts of the candidate sections the model was shown, and as many as it says were returned")
    else:
        slices = run.get("context_slices", [])
    if user is not None:
        for s in slices:
            section = sections.get(s["section_id"])
            if section is None:
                report(False, f"context slice {s['label']}: section missing from the pack")
                continue
            shown = section["text"][s["start"] : s["end"]]
            ok = sha256_text(shown) == s["sha256"] and shown in user
            report(ok, f"context slice {s['label']} (rank {s['rank']}): {s['end'] - s['start']} characters{' of a truncated section' if s['truncated'] else ''}, in the user message")

    recorded_unicode = (run.get("verifier") or {}).get("unicode")
    if recorded_unicode and recorded_unicode != unicodedata.unidata_version:
        print(f"NOTE the pack was made with Unicode {recorded_unicode}; this Python has {unicodedata.unidata_version}. A span holding a character this Python does not know is failed, not judged")
    v8 = version >= 8
    if v8:
        print(f"NOTE verified under {run['options']['verifier_version']}: spans are checked with its edges and its signed numbers")
    checked = withheld = 0
    for finding in findings:
        for span in finding["spans"]:
            label = f"finding {finding['ordinal'] + 1} span {span['ordinal'] + 1}"
            if not span["verified"]:
                withheld += 1
                print(f"WITHHELD {label}: the workbench did not find this quote; nothing asserted")
                continue
            section = sections.get(span["section_id"])
            if section is None:
                report(False, f"{label}: section {span['section_id']} missing from the pack")
                continue
            found = section["text"][span["start"] : span["end"]]
            checked += 1
            where = f"section {section['number'] or section['heading']} characters {span['start']}-{span['end']} ({span['method']})"
            if v8 and WRITTEN is None:
                report(False, f"{label}: {where}: a verifier v8 span needs Python 3.11 or later to be checked")
                continue
            unread = [c for c in found + span["quote"] if unicodedata.category(c) == "Cn"]
            if unread and recorded_unicode != unicodedata.unidata_version:
                report(False, f"{label}: {where}: holds characters this Python's Unicode {unicodedata.unidata_version} does not know; check it with a Python whose Unicode is {recorded_unicode} or later")
                continue
            if version >= 5 and window and span["end"] > window:
                # Since v5 the workbench searches only the window the prompt carried: a span past it is not one it found.
                report(False, f"{label}: {where}: ends beyond the {window} characters of its section the run searched")
                continue
            searched = section["text"][:window] if window else section["text"]
            in_place = (searched, span["start"], span["end"])
            label_text = f"{section['number']} {section['heading']}".strip()  # as the workbench named it (runs/service._label)
            edges_in = section["text"] if version >= 11 else searched  # v11: a window's end is not the text's end
            ok = same(found, span["quote"], span["method"], version, in_place, label_text) and (not v8 or edges_hold(edges_in, span["start"], span["end"], version))
            report(ok, f"{label}: {where}")
    shown_to_model = {s["section_id"]: s for s in slices}
    beyond = 0
    for finding in findings:
        for span in finding["spans"]:
            s = shown_to_model.get(span["section_id"])
            if span["verified"] and s is not None and span["end"] > s["end"]:
                beyond += 1
                print(f"NOTE finding {finding['ordinal'] + 1} span {span['ordinal'] + 1} lies beyond the {s['end']} characters of its section the model was shown")
    # The contract's own replacement and precedence statements the run recorded: each end is where the record says it is.
    # A PASS says the words are there; it says nothing about their legal effect.
    if run.get("relations_file"):
        relations = json.loads(text_of(here / run["relations_file"]))
        for index, relation in enumerate(relations):
            for end in ("statement", "replaced", "replacement"):
                passage = relation.get(end)
                if not passage:
                    continue
                section = sections.get(passage["section_id"])
                ok = section is not None and section["text"][passage["start"] : passage["end"]] == passage["text"]
                report(ok, f"relation {index + 1} ({relation['kind']}) {end}: {passage['label']} characters {passage['start']}-{passage['end']}")
    elif "relations_file" in run:
        print(f"NOTE {run.get('relations_note')}")

    if bound:
        # A person's review is a row of its own, written after the run; review_head names the latest review row of each
        # finding, so with none recorded it is worked out from the findings' ids alone, and an invented review fails.
        unreviewed = sha256_text(chr(31).join(f"{f['id']}:" for f in findings))
        if run.get("review_head") == unreviewed:
            report(all(f["review"] is None for f in findings), "reviews: none was recorded when the pack was made (review_head), and findings.json shows none")
        else:
            unchecked("findings.json review; run.json review_head", "a person's review is recorded after the run, and no hash covers what it says")
        report(holds(here, "README.txt", README_SHA256), "README.txt: the README this checker came with")
        unchecked(
            "run.json created_at, finished_at, packed_at, document.name, document.parser_version, document.parse_coverage, workbench_link, "
            "verifier.describes, verifier.version, verifier.ladder, verifier.label_stripping, verifier.boundary, verifier.unicode, relations_note, "
            "model_input.problem, model_input.model_calls, model_input.validation_rounds, stages.attempt, stages.rounds",
            "no hash the run recorded covers them: when it ran and was packed, the document's name and reading, the build that made this pack, "
            "how many calls the model took and why an input could not be rebuilt",
        )
        unchecked(
            "findings.json decided_by, flagged, concerns, code_compared, unseen_reason, spans.clause_label",
            "worked out from the fields above when this pack was made, by the build that made it; this file does not work them out again",
        )
    print(f"\n{checked} located span(s) checked, {withheld} withheld, {beyond} beyond the model-seen slice, {failures} failure(s), {left} line(s) of fields unchecked")
    return 1 if failures else 0


if __name__ == "__main__":
    sys.exit(main())
