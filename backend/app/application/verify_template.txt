#!/usr/bin/env python3
"""Verify this evidence pack on any machine with Python 3. No dependencies.

Checks, in order:
  1. the document file's sha256 is the one the run recorded;
  2. the canonical sections file's sha256 is the one the run recorded, and the guidance file's, when the
     run had guidance;
  3. every span the workbench marked verified is found again: the characters at the stored
     offsets of the stored section text match the quoted passage under the same ladder the
     workbench used (exact, normalized, casefold, typed tokens; letters-and-digits on records made
     before verifier v5; label stripping when the workbench recorded it). Since verifier v6 the workbench
     compares words whole; the typed check here joins neighbouring words, as v5 did, so it holds for
     records of both and is never stricter than what was recorded;
  4. spans the workbench marked unverified are listed, with nothing to check: the workbench
     withheld them;
  5. when the pack carries the rebuilt model input, its hash is the one the run recorded when it
     made the call, and every context slice (the part of a section the model was shown) hashes
     to what run.json says and appears verbatim in the user message;
  6. a located span that lies beyond the slice the model was shown is reported: the quote exists
     in the document, but the model never saw those characters (a NOTE, not a failure).
This verifier establishes integrity between the original input, the canonical reading, the context
given to the model and the stored spans. It does not, and cannot, say the conclusions are right.
Exit code 0 when every check passes, 1 otherwise.
"""
import decimal
import hashlib
import json
import pathlib
import re
import sys

HERE = pathlib.Path(__file__).resolve().parent
QUOTE_MAP = json.loads(__QUOTE_MAP__)


def sha256(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def normalize(text):
    out = []
    pending = False
    for ch in text:
        mapped = QUOTE_MAP.get(ch, ch)
        if mapped == "":
            continue
        if mapped.isspace():
            pending = True
            continue
        if pending and out:
            out.append(" ")
        pending = False
        out.append(mapped)
    return "".join(out)


def alnum(text):
    return "".join(ch.casefold() for ch in text if ch.isalnum())


TOKEN = re.compile(
    r"(?P<section>\d+(?:\.\d+){2,}(?:\([a-z0-9]+\))*|\d+(?:\.\d+)?(?:\([a-z0-9]+\))+)"
    r"|(?P<currency>(?:[$€£¥]|USD|EUR|GBP|CHF|CAD|AUD)\s?\d+(?:,\d+)*(?:\.\d+)?)"
    r"|(?P<percent>\d+(?:,\d+)*(?:\.\d+)?\s?(?:%|percent\b|per\s+cent\b))"
    r"|(?P<number>\d+(?:,\d+)*(?:\.\d+)?)"
    r"|(?P<word>[^\W\d_]+)"
    r"|(?P<punct>[^\w\s]|_)",
    re.IGNORECASE,
)


def number(text):
    compact = text.replace(" ", "")
    # A comma separates thousands only in thousands grouping: "1,5" and "15,00" are not 15 and 1500 (verifier v7).
    if "," in compact and not re.fullmatch(r"\d{1,3}(?:,\d{3})+(?:\.\d+)?", compact):
        return compact
    try:
        return format(decimal.Decimal(compact.replace(",", "")).normalize(), "f")
    except decimal.InvalidOperation:
        return text


def typed(text):
    """The workbench's typed tokens (verify/tokens.py) in standard-library form: rigid tokens and runs of joined words."""
    items = []
    run = []
    for match in TOKEN.finditer(text):
        kind = match.lastgroup
        surface = match.group(0)
        if kind == "punct":
            continue
        if kind == "word":
            run.append(surface.casefold())
            continue
        if run:
            items.append(("word", "".join(run)))
            run = []
        if kind == "number":
            items.append(("number", number(surface)))
        elif kind == "currency":
            symbol = re.match(r"(?:[$€£¥]|USD|EUR|GBP|CHF|CAD|AUD)", surface, re.IGNORECASE).group(0)
            items.append(("currency", symbol.upper() + number(surface[len(symbol):])))
        elif kind == "percent":
            items.append(("percent", number(re.match(r"[\d,]*(?:\.\d+)?", surface).group(0)) + "%"))
        else:
            items.append(("section", surface.lower()))
    if run:
        items.append(("word", "".join(run)))
    return items



def same(found, quote, method):
    base = method.split(":")[-1]
    if "unprefixed" in method:
        return bool(alnum(found)) and alnum(quote).endswith(alnum(found))
    if base == "exact":
        return found == quote
    if base == "normalized":
        return normalize(found) == normalize(quote).strip()
    if base == "casefold":
        return normalize(found).casefold() == normalize(quote).strip().casefold()
    if base == "alnum":
        return alnum(found) == alnum(quote)
    if base == "typed":
        return typed(found) == typed(quote)
    return False


def main():
    run = json.loads((HERE / "run.json").read_text(encoding="utf-8"))
    sections = {s["id"]: s for s in json.loads((HERE / "sections.json").read_text(encoding="utf-8"))}
    findings = json.loads((HERE / "findings.json").read_text(encoding="utf-8"))
    failures = 0

    def report(ok, label):
        nonlocal failures
        failures += 0 if ok else 1
        print(("PASS " if ok else "FAIL ") + label)

    document = run["document"]
    if document.get("file"):
        report(sha256(HERE / document["file"]) == document["sha256"], f"document bytes: sha256 {document['sha256'][:16]}...")
    else:
        print("NOTE the original document bytes are not in this pack; the sections hash below still ties the record to what was read")
    report(sha256(HERE / "sections.json") == run["sections_sha256"], f"canonical sections: sha256 {run['sections_sha256'][:16]}...")
    recorded = run.get("sections_text_sha256_recorded")
    if recorded:
        joined = chr(31).join(s["text"] for s in sorted(sections.values(), key=lambda s: s["ordinal"]))
        report(hashlib.sha256(joined.encode("utf-8")).hexdigest() == recorded, f"sections are what the run read: recorded {recorded[:16]}...")
    else:
        print("NOTE the run recorded no reading hash (older record); the sections hash above ties the pack to itself only")
    if run.get("guidance_file"):
        report(sha256(HERE / run["guidance_file"]) == run.get("guidance_sha256"), f"guidance: sha256 {str(run.get('guidance_sha256'))[:16]}...")

    checked = withheld = 0
    for finding in findings:
        for span in finding["spans"]:
            label = f"finding {finding['ordinal'] + 1} span {span['ordinal'] + 1}"
            if not span["verified"]:
                withheld += 1
                print(f"WITHHELD {label}: the workbench did not find this quote; nothing asserted")
                continue
            section = sections.get(span["section_id"])
            if section is None:
                report(False, f"{label}: section {span['section_id']} missing from the pack")
                continue
            found = section["text"][span["start"] : span["end"]]
            checked += 1
            where = f"section {section['number'] or section['heading']} characters {span['start']}-{span['end']} ({span['method']})"
            report(same(found, span["quote"], span["method"]), f"{label}: {where}")
    slices = {s["section_id"]: s for s in run.get("context_slices", [])}
    model_input = run.get("model_input") or {}
    if model_input.get("user_file") and model_input.get("system_file"):
        system = (HERE / model_input["system_file"]).read_text(encoding="utf-8")
        user = (HERE / model_input["user_file"]).read_text(encoding="utf-8")
        prompt_hex = hashlib.sha256(system.encode("utf-8")).hexdigest()
        recomputed = hashlib.sha256((prompt_hex + user).encode("utf-8")).hexdigest()
        report(recomputed == model_input.get("input_sha256_recorded"), f"model input: rebuilt messages hash to what the run recorded {str(model_input.get('input_sha256_recorded'))[:16]}...")
        report(prompt_hex == run["prompt_hash"], f"prompt {run['prompt_version']}: system text hashes to the recorded prompt hash")
        for s in run.get("context_slices", []):
            section = sections.get(s["section_id"])
            if section is None:
                report(False, f"context slice {s['label']}: section missing from the pack")
                continue
            shown = section["text"][s["start"] : s["end"]]
            ok = hashlib.sha256(shown.encode("utf-8")).hexdigest() == s["sha256"] and shown in user
            report(ok, f"context slice {s['label']} (rank {s['rank']}): {s['end'] - s['start']} characters{' of a truncated section' if s['truncated'] else ''}, in the user message")
    elif model_input:
        print(f"NOTE the model input could not be rebuilt from the record: {model_input.get('problem')}")
    beyond = 0
    for finding in findings:
        for span in finding["spans"]:
            s = slices.get(span["section_id"])
            if span["verified"] and s is not None and span["end"] > s["end"]:
                beyond += 1
                print(f"NOTE finding {finding['ordinal'] + 1} span {span['ordinal'] + 1} lies beyond the {s['end']} characters of its section the model was shown")
    print(f"\n{checked} located span(s) checked, {withheld} withheld, {beyond} beyond the model-seen slice, {failures} failure(s)")
    return 1 if failures else 0


if __name__ == "__main__":
    sys.exit(main())
